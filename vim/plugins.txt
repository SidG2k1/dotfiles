# vim/plugins.txt - the plugins vimrc requires, as vim native packages.
#
# One `owner/repo` per line; '#' starts a comment. Vim 8+ loads everything
# under ~/.vim/pack/bundle/start/ by itself (:help packages) - no plugin
# manager. Each plugin below backs a specific vimrc setting or mapping that
# silently does nothing without it.
#
# install.sh clones each line into ~/.vim/pack/bundle/start/<repo>. Then run
# `vim -es -c 'helptags ALL' -c q` for the plugin docs.
#
# Plugins are third-party checkouts and are deliberately not vendored into this
# repo; this file is the record of which ones vimrc expects.

# Async linting. Backs g:ale_lint_on_text_changed and the SR mapping (:ALEReset).
dense-analysis/ale

# Statusline. Backs the g:airline_section_z line/percent readout.
vim-airline/vim-airline

# Thesaurus lookup. Backs the ;th mapping
# (:ThesaurusQueryReplaceCurrentWord) used while writing prose and LaTeX.
ron89/thesaurus_query.vim

# Motion jumps (<leader><leader>w, <leader><leader>f and friends) with mapleader
# set to space.
easymotion/vim-easymotion

# :Tabularize alignment for tables and assignment blocks, including markdown and
# LaTeX tabular environments written with the ;beg snippets.
godlygeek/tabular

# Git from inside vim - :Git, :Gdiffsplit, :Gblame. Note the shell's git already
# routes through delta; fugitive does its own rendering.
tpope/vim-fugitive

# Sign-column diff markers against the index, alongside vimrc's number /
# relativenumber gutter.
airblade/vim-gitgutter

# Jsonnet filetype, syntax, and indent support. No jsonnet handling exists in
# vim without it.
google/vim-jsonnet
